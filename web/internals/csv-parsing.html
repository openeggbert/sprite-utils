CSV Parsing

Source: src/SpriteSheet.cpp, src/SpriteSheetRow.cpp

SpriteSheet Constructor Flow

SpriteSheet(path)
  1. text = Utils::readTextFromFile(path)
  2. Create istringstream from text
  3. getline → skip header line
  4. Loop: getline each line
     if line contains "skipskip" → break
     processLine(line, rows)
  5. saveComputedFile(path, text, rows)

processLine(line, rows)

void SpriteSheet::processLine(const string& line, vector<SpriteSheetRow>& rows)
  1. Parse SpriteSheetRow from line (CSV split by ";")
  2. validateRow(row)         // validate file context
  3. Height negation logic:   // multi-column rows
     if column > 1 AND previous exists
        AND abs(height) >= abs(prev.height)
        AND (column > 2 ? both non-zero heights : false)
       → row.height = -row.height
  4. updateSpriteSheetRow(row) // compute missing x and height
  5. rows.push_back(row)
  6. updateMap(row)
  7. Print row to stdout

SpriteSheetRow Constructor (CSV parsing)

SpriteSheetRow(csvLine)
  cols = Utils::split(csvLine, ";")
  if cols.size() < 10 → throw invalid_argument
  file     = cols[0]
  group    = cols[1]
  numberInGroup = stoi(cols[2])
  row      = stoi(cols[3])
  column   = stoi(cols[4])
  x        = parseOptionalInt(cols[5]).value_or(-1)  // -1 = compute later
  y        = stoi(cols[6])
  width    = stoi(cols[7])
  height   = (empty or "0") ? 0 : stoi(cols[8])
  notes    = cols[9] if exists else ""
  tags     = cols[10] if exists else ""
  numberPerSheet = stoi(cols[11]) if exists else 0
  if column > 1: height = -height  // encode as negative for multi-column

updateSpriteSheetRow(row)

if row.x == -1:
    if column == 1: row.x = 0
    else:           row.x = lastSpriteSheetRow.x + lastSpriteSheetRow.width

if row.height <= 0:
    row.height = lastSpriteSheetRow.height + (-row.height)
    // (-row.height) = delta from previous height

lastSpriteSheetRow = row  // save as static for next iteration

validateRow(row)

if lastSpriteSheetRow exists AND row.file != lastSpriteSheetRow.file:
    lastSpriteSheetRow.reset()  // new file context

if !lastSpriteSheetRow:
    validateFirstRow(row)       // must be row==1, col==1
else:
    validateSubsequentRow(row)  // must be sequential

validateFirstRow

if row.row != 1 OR row.column != 1:
    throw SpriteUtilsException("Invalid initial row or column for file " + file)
row.numberPerSheet = 1

validateSubsequentRow

row.numberPerSheet = lastSpriteSheetRow.numberPerSheet + 1
if row.row > lastSpriteSheetRow.row:
    if row.row != lastSpriteSheetRow.row + 1: throw  // non-sequential row jump
elif row.row < lastSpriteSheetRow.row:
    throw  // going backwards
else (same row):
    if row.column != lastSpriteSheetRow.column + 1: throw  // non-sequential column

Height Encoding Convention

In the CSV, for column > 1, the height field is initially negated by the constructor (if column > 1: height = -height). Then in processLine, if the negated height (absolute) is ≥ the previous height and column > 2, it is negated again. This double-negation mechanism allows encoding height as a delta relative to the previous row. It is non-obvious and represents a compact encoding chosen for the CSV format.
